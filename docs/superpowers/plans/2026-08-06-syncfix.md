# Auto-Resolve Conflicts on Fix PRs — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A daily scheduled job finds open `codesentinel/fix-*` PRs blocked by merge conflicts, resolves them by merging `main` and re-applying AI fixes, then enables native GitHub auto-merge so the PR merges when checks pass.

**Architecture:** New engine mode `syncfix` reuses the existing fix pipeline (`runFix`, `pushFixes`) scoped to conflicted files via a new `files` config option. New `GitHubReporter` methods handle PR listing, state, native auto-merge, and direct merge. A new workflow `fix-pr-sync.yml` runs the mode once daily.

**Tech Stack:** TypeScript (strict, ESM), Node >=18, `node:child_process` execFileSync for git, `node:util` parseArgs, vitest, GitHub REST API via fetch.

## Global Constraints

- All code strict TypeScript, ESM (`"type": "module"`), imports end in `.js`.
- Do NOT add comments to code unless the surrounding code has them (match style).
- Only `codesentinel/fix-*` branches are ever modified; never force-push (`--force`).
- New mode name: `syncfix`. Config keys: `files` (string[]), `maxSyncfixPRs` (number, default 1).
- Tests live in `tests/`, use vitest with `globals: true`, inline fixtures, no network.
- Verify with `npm run typecheck`, `npm test`, `npm run build` before committing each task.

```ts
  /** Enable auto-merge on a PR (merges when all required checks pass). */
  async enableAutoMerge(pullNumber: number, mergeMethod: "merge" | "squash" | "rebase" = "squash"): Promise<void> {
    await this.mergePR(pullNumber, mergeMethod);
  }

  /** List all open pull requests (up to 100). */
  async listOpenPRs(): Promise<OpenPR[]> {
    const url = `${this.api}/repos/${this.coords.owner}/${this.coords.repo}/pulls?state=open&per_page=100`;
    const result = await this.request("GET", url) as OpenPR[] | null;
    return result ?? [];
  }

  /** Get a single PR including mergeable state. */
  async getPR(pullNumber: number): Promise<OpenPR> {
    const url = `${this.api}/repos/${this.coords.owner}/${this.coords.repo}/pulls/${pullNumber}`;
    return await this.request("GET", url) as OpenPR;
  }

  /** Enable GitHub native auto-merge (merges when all checks pass). */
  async enableNativeAutoMerge(pullNumber: number, mergeMethod: "merge" | "squash" | "rebase" = "squash"): Promise<boolean> {
    try {
      const url = `${this.api}/repos/${this.coords.owner}/${this.coords.repo}/pulls/${pullNumber}`;
      await this.request("PATCH", url, { auto_merge: { enabled: true, merge_method: mergeMethod } });
      return true;
    } catch (err) {
      logger.warn(`enableNativeAutoMerge: failed for #${pullNumber}: ${err instanceof Error ? err.message : err}`);
      return false;
    }
  }

  /** Directly merge a PR (only succeeds when checks pass / no conflicts). */
  async mergePR(pullNumber: number, mergeMethod: "merge" | "squash" | "rebase" = "squash"): Promise<boolean> {
    try {
      const url = `${this.api}/repos/${this.coords.owner}/${this.coords.repo}/pulls/${pullNumber}/merge`;
      await this.request("PUT", url, { merge_method: mergeMethod });
      return true;
    } catch (err) {
      logger.warn(`mergePR: failed for #${pullNumber}: ${err instanceof Error ? err.message : err}`);
      return false;
    }
  }
```

```ts
  /** syncfix: resolve conflicts on open codesentinel/fix-* PRs, then enable auto-merge. */
  private async runSyncFix(): Promise<EngineReport> {
    const { execFileSync } = await import("node:child_process");
    const baseReport = {
      mode: "syncfix" as const,
      findings: [] as never[],
      score: null,
      comments: [] as never[],
      generatedTests: [] as never[],
      fixAttempts: [] as never[],
      metrics: { filesAnalyzed: 0, findingsBySeverity: {} as Record<string, number>, durationMs: 0 },
    };

    const token = this.secrets.github_token ?? process.env.GITHUB_TOKEN;
    const repo = process.env.GITHUB_REPOSITORY;
    if (!token || !repo) {
      const summary = "syncfix: missing GITHUB_TOKEN or GITHUB_REPOSITORY — nothing to do";
      logger.warn(summary);
      return { ...baseReport, summary };
    }
    const [owner, repoName] = repo.split("/");
    if (!owner || !repoName) {
      const summary = `syncfix: invalid GITHUB_REPOSITORY '${repo}'`;
      logger.warn(summary);
      return { ...baseReport, summary };
    }

    const reporter = new GitHubReporter({ token, owner, repo: repoName });
    const prs = await reporter.listOpenPRs();
    const fixPRs = prs.filter(
      (p) => p.head.ref.startsWith("codesentinel/fix-") && p.mergeable === false,
    );
    logger.info(`syncfix: ${fixPRs.length} conflicted fix PR(s) of ${prs.length} open`);

    let resolved = 0;
    for (const pr of fixPRs.slice(0, this.config.maxSyncfixPRs ?? 1)) {
      const ok = await this.resolveConflictedPR(pr, reporter);
      if (!ok) continue;
      resolved++;
      const updated = await reporter.getPR(pr.number);
      const autoMerged = await reporter.enableNativeAutoMerge(pr.number, "squash");
      if (!autoMerged && updated.mergeable_state === "clean") {
        await reporter.mergePR(pr.number, "squash");
      }
    }

    const summary = `syncfix: resolved ${resolved} PR(s) of ${fixPRs.length} conflicted`;
    logger.info(summary);
    return { ...baseReport, summary };
  }

  /** Merge base into a fix branch, keep main's version of conflicted files, re-apply AI fixes. */
  private async resolveConflictedPR(pr: OpenPR, _reporter: GitHubReporter): Promise<boolean> {
    const { execFileSync } = await import("node:child_process");
    const branch = pr.head.ref;
    const base = pr.base.ref;
    try {
      execFileSync("git", ["config", "user.email", "bot@codesentinel.ai"], { cwd: this.root, stdio: "pipe" });
      execFileSync("git", ["config", "user.name", "CodeSentinel Bot"], { cwd: this.root, stdio: "pipe" });
      execFileSync("git", ["fetch", "origin", branch], { cwd: this.root, stdio: "pipe", timeout: 60000 });
      execFileSync("git", ["checkout", "-B", branch, `origin/${branch}`], { cwd: this.root, stdio: "pipe" });

      let mergeOk = true;
      try {
        execFileSync("git", ["merge", `origin/${base}`], { cwd: this.root, stdio: "pipe", timeout: 60000 });
      } catch {
        mergeOk = false;
      }

      let conflicted: string[] = [];
      try {
        const out = execFileSync("git", ["diff", "--name-only", "--diff-filter=U"], { cwd: this.root, stdio: "pipe" }).toString();
        conflicted = out.split("\n").map((s) => s.trim()).filter(Boolean);
      } catch {
        // no unmerged paths
      }

      if (!mergeOk && conflicted.length === 0) {
        logger.warn(`syncfix: merge of ${branch} onto ${base} failed for unknown reason — skipping`);
        return false;
      }

      if (conflicted.length > 0) {
        logger.info(`syncfix: resolving ${conflicted.length} conflicted file(s) on ${branch}`);
        for (const f of conflicted) {
          execFileSync("git", ["checkout", "--theirs", "--", f], { cwd: this.root, stdio: "pipe" });
        }
        execFileSync("git", ["add", "-A"], { cwd: this.root, stdio: "pipe" });
        execFileSync("git", ["commit", "-m", `Merge ${base} into ${branch} [skip ci]`], { cwd: this.root, stdio: "pipe" });

        const prevFiles = this.config.files;
        const prevRefName = process.env.GITHUB_REF_NAME;
        this.config.files = conflicted;
        process.env.GITHUB_REF_NAME = branch;
        try {
          await this.runFix();
        } finally {
          this.config.files = prevFiles;
          if (prevRefName === undefined) delete process.env.GITHUB_REF_NAME;
          else process.env.GITHUB_REF_NAME = prevRefName;
        }
      } else if (!mergeOk) {
        logger.info(`syncfix: no conflicts on ${branch}, nothing to do`);
      }
      return true;
    } catch (err) {
```
